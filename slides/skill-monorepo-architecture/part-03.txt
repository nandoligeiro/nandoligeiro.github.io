const deck=document.getElementById('deck');
const progress=document.getElementById('progress');
const sections=[...document.querySelectorAll('section')];
let i=0;
function go(n){
  i=Math.max(0,Math.min(sections.length-1,n));
  deck.style.transform=`translateX(${-i*100}vw)`;
  progress.style.width=`${((i+1)/sections.length)*100}%`;
  location.hash=`slide-${i+1}`;
}
window.addEventListener('keydown',e=>{
  if(['ArrowRight','PageDown',' '].includes(e.key)) go(i+1);
  if(['ArrowLeft','PageUp'].includes(e.key)) go(i-1);
  if(e.key==='Home') go(0);
  if(e.key==='End') go(sections.length-1);
  if(e.key.toLowerCase()==='n') document.body.classList.toggle('show-notes');
});
let lock=false;
window.addEventListener('wheel',e=>{
  if(lock) return;
  if(Math.abs(e.deltaY)>35 || Math.abs(e.deltaX)>35){
    lock=true;
    go(i + ((e.deltaY>0 || e.deltaX>0)?1:-1));
    setTimeout(()=>lock=false,430);
  }
},{passive:true});
const m=(location.hash||'').match(/slide-(\d+)/);
go(m?Number(m[1])-1:0);
</script>
